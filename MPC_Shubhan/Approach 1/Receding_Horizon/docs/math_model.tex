This document is the formal companion to the code in \texttt{Receding\_Horizon/code/}
(the window MILP is built in \texttt{4\_MCSModel.jl}; the closed loop in
\texttt{5\_MPCLoop.jl}). Plain-English descriptions and the full data schema are in
\texttt{README.md}; a line-by-line code audit is in
\texttt{constraints\_code\_vs\_model.txt}.

The controller is a \textbf{multi-day, cross-day receding horizon}. Each 15-minute step
re-solves a MILP over a window that spans the rest of the current day plus
\texttt{lookahead\_days} future daytime blocks; only the first interval is applied. Days
are chained: the MCS battery is recharged overnight, while the CEV battery and any
unfinished work carry over.

%======================================================================
\section{Sets and indices}
\begin{tabular}{ll}
$m \in \mathcal{M}$ & mobile charging stations (MCS) \\
$e \in \mathcal{E}$ & construction EVs (CEV / excavators) \\
$i,j \in \mathcal{N}$ & nodes; $\mathcal{N}_g$ grid nodes, $\mathcal{N}_c$ construction sites \\
$a \in \mathcal{B}$ & activities $=\{\text{dig},\ \text{load},\ \text{trv},\ \text{idle}\}$ \\
$d \in \mathcal{D}$ & simulated days $\{1,\dots,n_{\text{days}}+1\}$ (day $n_{\text{days}}+1$ = dropped buffer) \\
$k$ & GLOBAL daytime interval, $k=(d-1)\,n_{\text{day}}+w$, $\ w=\mathrm{wd}(k)\in\{1,\dots,n_{\text{day}}\}$ \\
$K_{\text{win}}$ & the intervals of the current re-solve window \\
$\mathcal{T}_b$ & interval boundaries of the window ($|K_{\text{win}}|+1$ of them) \\
\end{tabular}

\medskip
$n_{\text{day}}$ (daytime intervals per day, e.g.\ 40 $=$ 08:00--18:00 at $\dt=0.25$\,h)
is \emph{inferred}: the last interval with nonzero work availability plus a 1-hour return
buffer. $A_{i,e}=1$ iff CEV $e$ works at site $i$.

\begin{tabular}{ll}

$\overline{SOE}^{\text{MCS}}_m,\underline{SOE}^{\text{MCS}}_m,SOE^{\text{MCS,ini}}_m$ & MCS energy cap / floor / start \\

$CH^{\text{CEV}}_e$ & CEV charge-acceptance rate \\

\section{Decision variables}
\textbf{Continuous ($\ge 0$):}
$P^{\text{ch}}_{m,i,k}$ (grid$\to$MCS), $P^{\text{dch}}_{m,i,k}$, $P^{\text{MCS-CEV}}_{m,i,e,k}$,
$P^{\text{work}}_{i,e,k}$, totals $P^{\text{ch,tot}}_{m,k},P^{\text{dch,tot}}_{m,k}$;
travel energy $L_{m,i,j,k}$; state $SOE^{\text{MCS}}_{m,t},SOE^{\text{CEV}}_{e,t}$ on
boundaries $t\in\mathcal{T}_b$; peaks $P^{\text{NC}},P^{\text{OP}}$; slacks
$s^{\text{miss,dig}}_{i,d},s^{\text{miss,load}}_{i,d},s^{\text{prec}}_{i,k},
s^{\text{pace}\pm}_{e,k},s^{\text{term}}_{e,d}$.

\smallskip
\textbf{Binary:} $u_{e,i,a,k}$ (activity), $\mu_{i,e,k}$ (CEV charging),
$\rho_{m,i,e,k}$ (plugged), $z_{m,i,k}$ (parked), $x_{m,i,j,k}$ (departs $i\!\to\!j$),
$y_{m,i,j,k}$ (in transit), $\beta^{\text{arr}},\beta^{\text{dep}}$ (arrival/departure).

%======================================================================
\section{Objective (Eq.\ 1)}
\begin{align}
\min\ &\sum_{m,k}\lambda^{\text{buy}}_{\mathrm{wd}(k)}P^{\text{ch,tot}}_{m,k}\dt
      +\sum_{m,k}\tfrac{c_{\text{ton}}}{1000}\lambda^{\text{CO2}}_{\mathrm{wd}(k)}P^{\text{ch,tot}}_{m,k}\dt
      \notag\\
     &+\lambda^{\text{NC}}P^{\text{NC}}+\lambda^{\text{OP}}P^{\text{OP}}
      +\rho_{\text{miss}}\!\!\sum_{i\in\mathcal{N}_c,d}\!\!\big(s^{\text{miss,dig}}_{i,d}+s^{\text{miss,load}}_{i,d}\big)
      +\rho_{\text{lab}}\dt\!\!\sum_{m,i,j,k}\!\! y_{m,i,j,k}.
\end{align}
(Precedence, pacing and terminal slacks are pinned to $0$ unless the corresponding
\texttt{soft\_*} flag is set.)

%======================================================================
\section{Constraints}

\paragraph{Power aggregation \& flow (HARD).}
\begin{align}
P^{\text{ch,tot}}_{m,k}&=\textstyle\sum_{i\in\mathcal{N}_g}P^{\text{ch}}_{m,i,k},\quad
P^{\text{dch,tot}}_{m,k}=\textstyle\sum_{i\in\mathcal{N}_c}P^{\text{dch}}_{m,i,k},\\
P^{\text{dch}}_{m,i,k}&=\textstyle\sum_{e}P^{\text{MCS-CEV}}_{m,i,e,k}\le DCH_m\,z_{m,i,k},\qquad
P^{\text{MCS-CEV}}_{m,i,e,k}\le DCH^{\text{plug}}_m\,\rho_{m,i,e,k}.
\end{align}
Grid charge only at grid nodes; site discharge only at sites.

\paragraph{State of energy (HARD).}
\begin{align}
SOE^{\text{MCS}}_{m,k+1}&=SOE^{\text{MCS}}_{m,k}+\eta_m P^{\text{ch,tot}}_{m,k}\dt
   -\tfrac{1}{\eta_m}P^{\text{dch,tot}}_{m,k}\dt-L^{\text{tot}}_{m,k},\\
SOE^{\text{CEV}}_{e,k+1}&=SOE^{\text{CEV}}_{e,k}
   +\textstyle\sum_{m,i}P^{\text{MCS-CEV}}_{m,i,e,k}\dt-\sum_i P^{\text{work}}_{i,e,k}\dt,\\
\underline{SOE}_\bullet &\le SOE_\bullet \le \overline{SOE}_\bullet .
\end{align}
\emph{Night link:} at each night boundary $SOE^{\text{MCS}}_{m}\!\leftarrow\!SOE^{\text{MCS,ini}}_m$
(recharged overnight); the CEV SOE carries over unchanged.

\paragraph{Plugging, presence \& routing (HARD).}
Charging implies idling, $\mu_{i,e,k}\le u_{e,i,\text{idle},k}$; at most $C^{\text{plug}}_m$
CEVs plugged. Each interval the MCS is parked at exactly one node or in transit on exactly
one arc:
$\sum_i z_{m,i,k}+\sum_{i\ne j}y_{m,i,j,k}=1$. Departure/arrival indicators and flow
conservation link $z_{m,i,k}$ across time; a trip on $i\!\to\!j$ lasts $\tau_{i,j}$
intervals and costs $L=k_{\text{trv}}$ (travel energy). \textbf{(10e)} at every evening
(each 18:00) the MCS is parked at a grid node: $\sum_{i\in\mathcal{N}_g}z_{m,i,\text{eve}}=1$.

\paragraph{Activity scheduling (HARD).}
Exactly one activity per assigned CEV per interval, capped by availability:
\begin{equation}
\textstyle\sum_a u_{e,i,a,k}=A_{i,e},\qquad
\sum_{a\in\{\text{dig,load,trv}\}}p_a\,u_{e,i,a,k}\le R_{i,e,\mathrm{wd}(k)}A_{i,e}(1-\mu_{i,e,k}).
\end{equation}

\paragraph{Per-day work quota (Eq.\ 12c) --- pinned BOTH ways.}
Let $Q^{\text{dig}}_{d,i}$ be the cumulative dig target through day $d$ (window-start
remaining $+\!\sum$ of each later day's fresh quota), and
$W^{\text{dig}}_{d,i}=\dt\sum_{e,\,k\le \text{end}(d)}u_{e,i,\text{dig},k}$ the cumulative dig
done. Then
\begin{align}
\text{(SOFT lower)}\quad & s^{\text{miss,dig}}_{i,d}\ \ge\ Q^{\text{dig}}_{d,i}-W^{\text{dig}}_{d,i},\\
\text{(HARD upper, no working ahead)}\quad & W^{\text{dig}}_{d,i}\ \le\ Q^{\text{dig}}_{d,i},
\end{align}
and identically for loading. The lower slack rolls over (penalised); the hard cap forbids
borrowing a future day's work while still allowing catch-up of earlier leftover.

\paragraph{Precedence (Eq.\ 12d, HARD).}
Within each day, cumulative loading $\le \text{scale}\times$ cumulative digging (dig before load):
\begin{equation}
\dt\!\!\sum_{e,\tau\le k}\!\!u_{e,i,\text{load},\tau}\ \le\
\text{scale}\;\dt\!\!\sum_{e,\tau\le k}\!\!u_{e,i,\text{dig},\tau}+s^{\text{prec}}_{i,k}.
\end{equation}

\paragraph{Rest rule (Eq.\ 12e, HARD).}
Over any rolling $(t_{\text{rest}}+\dt)$ window a CEV does at most $t_{\text{rest}}$ hours of
construction work (dig/load/trv), i.e.\ $\ge 1$ idle break. With
$r=\lceil t_{\text{rest}}/\dt\rceil$,
\begin{equation}
\textstyle\sum_{k=k_0}^{k_0+r}\sum_{a\in\{\text{dig,load,trv}\}}u_{e,i,a,k}\ \le\ r .
\end{equation}
The rule is \emph{seeded from the applied Work/Break history of the current day}, adding
``seam'' constraints so a work-run cannot leak across the every-15-min re-solves or across
day-end.

\paragraph{Travel pacing (Eq.\ 13, HARD).}
Cumulative traveling tied to cumulative productive work in a two-sided band
($\sim$1 travel per $\kappa$ productive intervals), with slacks $s^{\text{pace}\pm}$.

\paragraph{Daily CEV energy neutrality (Eq.\ 8b, HARD).}
At \emph{every} evening boundary (each 18:00) each CEV returns to its start-of-day level:
\begin{equation}
SOE^{\text{CEV}}_{e,\text{eve}+1}\ \ge\ SOE^{\text{CEV,ini}}_e-\varepsilon .
\end{equation}
Hence each reported day is CEV-energy-neutral (start $\approx$ end).

\paragraph{Demand peaks (E1).}
$P^{\text{NC}}\ge\sum_m P^{\text{ch,tot}}_{m,k}$ (all $k$) and
$P^{\text{OP}}\ge\sum_m P^{\text{ch,tot}}_{m,k}$ (on-peak $k$).

\paragraph{Keep-up reserve (E7).}
To make the daily terminal \emph{recursively} feasible, a per-boundary lower bound is
placed on each CEV's SOE, built backward from each day's 18:00 using the net charge rate
(charge minus the persistent idle draw while plugged in) and the last interval the MCS may
charge before leaving to be home by 18:00. Let $L_c$ be that last-charge interval,
$g^{\text{net}}$ the net per-interval gain, $\delta_{\text{idle}}$ the idle drain, and
$S^\star$ the SOE required at the idle-tail start; then for boundary $t$ within the day
\begin{equation}
SOE^{\text{CEV}}_{e,t}\ \ge\
\begin{cases}
S^\star-g^{\text{net}}\,\big((L_c+1)-t\big), & t\le L_c+1,\\
(SOE^{\text{CEV,ini}}_e-\varepsilon)+\delta_{\text{idle}}\,(\text{eve}+1-t), & t> L_c+1,
\end{cases}
\end{equation}
clamped to $[\underline{SOE},\overline{SOE}]$. It binds only in each day's late tail.

%======================================================================
\section{Two-phase / overnight}
Phase 1 is the daytime MILP above (08:00--18:00). Phase 2 (18:00--08:00) is a
\emph{deterministic} cheapest-hours recharge that returns the MCS to $SOE^{\text{MCS,ini}}$;
it is not part of the MILP and is modelled in the window by the night reset above.
