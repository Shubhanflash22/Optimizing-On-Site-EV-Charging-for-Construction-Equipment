Formal companion to the code in \texttt{Shrinking\_Horizon/code/} (the window MILP is
built in \texttt{4\_MCSModel.jl}; the closed loop in \texttt{5\_MPCLoop.jl}). Plain-English
descriptions and the data schema are in \texttt{README.md}; a code audit is in
\texttt{constraints\_code\_vs\_model.txt}.

The controller is a \textbf{single-day shrinking horizon}. Each 15-minute step re-solves a
MILP over the \emph{entire remaining daytime} $[k_0,\dots,n_{\text{day}}]$; only the first
interval is applied, and the window shrinks as the day advances. The night is handled by a
deterministic overnight recharge (Phase 2).

%======================================================================
\section{Sets and indices}
\begin{tabular}{ll}
$m \in \mathcal{M}$ & mobile charging stations (MCS) \\
$e \in \mathcal{E}$ & construction EVs (excavators) \\
$i,j \in \mathcal{N}$ & nodes; $\mathcal{N}_g$ grid, $\mathcal{N}_c$ sites \\
$a \in \mathcal{B}$ & activities $=\{\text{dig},\text{load},\text{trv},\text{idle}\}$ \\
$k$ & daytime interval $1,\dots,n_{\text{day}}$; window $[k_0,n_{\text{day}}]$ \\
$\mathcal{T}_b$ & interval boundaries of the window \\
\end{tabular}

\medskip
$n_{\text{day}}$ (e.g.\ 40 $=$ 08:00--18:00 at $\dt=0.25$\,h) is \emph{inferred}: last
work-available interval $+$ a 1-hour return buffer. $A_{i,e}=1$ iff CEV $e$ works at site
$i$.

%======================================================================
\section{Parameters (selected)}
\begin{tabular}{ll}
$\dt$ & interval length (h), $0.25$ \\
$\lambda^{\text{buy}}_{k},\lambda^{\text{CO2}}_{k}$ & price (\$/kWh) / carbon intensity \\
$\overline{SOE}_\bullet,\underline{SOE}_\bullet,SOE^{\text{ini}}_\bullet$ & energy cap / floor / start (MCS, CEV) \\
$CH_m,DCH_m,DCH^{\text{plug}}_m,C^{\text{plug}}_m,\eta_m,CH^{\text{CEV}}_e$ & MCS / CEV charge-discharge parameters \\
$p_a$ & activity power (kW); estimated online, MILP uses posterior mean \\
$H^{\text{dig}}_i,H^{\text{load}}_i$ & lumpsum dig / load requirement (hours) at site $i$ \\
$\tau_{i,j},k_{\text{trv}},R_{i,e,k}$ & travel time / energy / work availability \\
$\rho_{\text{miss}},\rho_{\text{lab}},\lambda^{\text{NC}},\lambda^{\text{OP}}$ & penalties \\
$\text{scale},\kappa,t_{\text{rest}},\varepsilon$ & precedence / pacing / rest / terminal tol. \\
\end{tabular}

\section{Decision variables}
\textbf{Continuous ($\ge0$):} $P^{\text{ch}}_{m,i,k}$, $P^{\text{dch}}_{m,i,k}$,
$P^{\text{MCS-CEV}}_{m,i,e,k}$, $P^{\text{work}}_{i,e,k}$, totals
$P^{\text{ch,tot}}_{m,k},P^{\text{dch,tot}}_{m,k}$, travel energy $L_{m,i,j,k}$, state
$SOE^{\text{MCS}}_{m,t},SOE^{\text{CEV}}_{e,t}$, peaks $P^{\text{NC}},P^{\text{OP}}$, slacks
$s^{\text{miss,dig}}_i,s^{\text{miss,load}}_i,s^{\text{prec}}_{i,k},s^{\text{pace}\pm}_{e,k},s^{\text{term}}_e$.

\smallskip
\textbf{Binary:} $u_{e,i,a,k}$, $\mu_{i,e,k}$, $\rho_{m,i,e,k}$, $z_{m,i,k}$,
$x_{m,i,j,k}$, $y_{m,i,j,k}$, $\beta^{\text{arr}},\beta^{\text{dep}}$.

%======================================================================
\section{Objective (Eq.\ 1)}
\begin{align}
\min\ &\sum_{m,k}\lambda^{\text{buy}}_{k}P^{\text{ch,tot}}_{m,k}\dt
      +\sum_{m,k}\tfrac{c_{\text{ton}}}{1000}\lambda^{\text{CO2}}_{k}P^{\text{ch,tot}}_{m,k}\dt
      +\lambda^{\text{NC}}P^{\text{NC}}+\lambda^{\text{OP}}P^{\text{OP}}\notag\\
     &+\rho_{\text{miss}}\!\!\sum_{i\in\mathcal{N}_c}\!\!\big(s^{\text{miss,dig}}_{i}+s^{\text{miss,load}}_{i}\big)
      +\rho_{\text{lab}}\dt\!\!\sum_{m,i,j,k}\!\! y_{m,i,j,k}.
\end{align}

%======================================================================
\section{Constraints}

\paragraph{Power aggregation \& flow (HARD).}
$P^{\text{ch,tot}}_{m,k}=\sum_{i\in\mathcal{N}_g}P^{\text{ch}}_{m,i,k}$,
$P^{\text{dch,tot}}_{m,k}=\sum_{i\in\mathcal{N}_c}P^{\text{dch}}_{m,i,k}$;
$P^{\text{dch}}_{m,i,k}=\sum_e P^{\text{MCS-CEV}}_{m,i,e,k}\le DCH_m z_{m,i,k}$;
$P^{\text{MCS-CEV}}_{m,i,e,k}\le DCH^{\text{plug}}_m\rho_{m,i,e,k}$.

\paragraph{State of energy (HARD).}
\begin{align}
SOE^{\text{MCS}}_{m,k+1}&=SOE^{\text{MCS}}_{m,k}+\eta_m P^{\text{ch,tot}}_{m,k}\dt-\tfrac{1}{\eta_m}P^{\text{dch,tot}}_{m,k}\dt-L^{\text{tot}}_{m,k},\\
SOE^{\text{CEV}}_{e,k+1}&=SOE^{\text{CEV}}_{e,k}+\textstyle\sum_{m,i}P^{\text{MCS-CEV}}_{m,i,e,k}\dt-\sum_i P^{\text{work}}_{i,e,k}\dt,\\
\underline{SOE}_\bullet&\le SOE_\bullet\le\overline{SOE}_\bullet .
\end{align}

\paragraph{Plugging, presence \& routing (HARD).}
$\mu_{i,e,k}\le u_{e,i,\text{idle},k}$; at most $C^{\text{plug}}_m$ plugged. Presence
partition $\sum_i z_{m,i,k}+\sum_{i\ne j}y_{m,i,j,k}=1$; departure/arrival + flow
conservation; trip on $i\!\to\!j$ lasts $\tau_{i,j}$ and costs $L=k_{\text{trv}}$.
\textbf{(10e)} MCS parked at a grid node by 18:00:
$\sum_{i\in\mathcal{N}_g}z_{m,i,n_{\text{day}}}=1$.

\paragraph{Activity scheduling (HARD).}
$\sum_a u_{e,i,a,k}=A_{i,e}$;
$\sum_{a\in\{\text{dig,load,trv}\}}p_a u_{e,i,a,k}\le R_{i,e,k}A_{i,e}(1-\mu_{i,e,k})$.

\paragraph{Work quota (Eq.\ 12c, SOFT) --- lumpsum.}
A single dig/load requirement per site; shortfall is penalised:
\begin{equation}
s^{\text{miss,dig}}_i\ \ge\ H^{\text{dig}}_i-\dt\!\!\sum_{e,k}\!u_{e,i,\text{dig},k},\qquad
s^{\text{miss,load}}_i\ \ge\ H^{\text{load}}_i-\dt\!\!\sum_{e,k}\!u_{e,i,\text{load},k}.
\end{equation}
(There is no hard upper cap: with a single day and daily neutrality there is no incentive
to overshoot.)

\paragraph{Precedence (Eq.\ 12d, HARD).}
$\dt\sum_{e,\tau\le k}u_{e,i,\text{load},\tau}\le\text{scale}\,\dt\sum_{e,\tau\le k}u_{e,i,\text{dig},\tau}+s^{\text{prec}}_{i,k}$.

\paragraph{Rest rule (Eq.\ 12e, HARD).}
With $r=\lceil t_{\text{rest}}/\dt\rceil$,
$\sum_{k=k_0}^{k_0+r}\sum_{a\in\{\text{dig,load,trv}\}}u_{e,i,a,k}\le r$, \emph{seeded from
the applied Work/Break history} so a work-run cannot leak across the every-15-min
re-solves or at day-end.

\paragraph{Travel pacing (Eq.\ 13, HARD).}
Cumulative traveling tied to cumulative productive work in a two-sided band ($\sim$1 travel
per $\kappa$ productive intervals), with slacks $s^{\text{pace}\pm}$.

\paragraph{CEV energy neutrality (Eq.\ 8b, HARD).}
$SOE^{\text{CEV}}_{e,n_{\text{day}}+1}\ \ge\ SOE^{\text{CEV,ini}}_e-\varepsilon$.

\paragraph{Demand peaks (E1) \& keep-up reserve (E7).}
$P^{\text{NC}}\!\ge\!\sum_m P^{\text{ch,tot}}_{m,k}$ (all $k$); $P^{\text{OP}}$ on-peak $k$.
E7: a per-boundary lower bound on $SOE^{\text{CEV}}_{e,t}$, built backward from 18:00 using
the net charge rate $g^{\text{net}}=\min(CH^{\text{CEV}}_e,DCH^{\text{plug}})\dt-\delta_{\text{idle}}$
and the last interval $L_c=n_{\text{day}}-\tau_{\text{grid}}$ the MCS may charge before
leaving:
\begin{equation}
SOE^{\text{CEV}}_{e,t}\ \ge\
\begin{cases}
S^\star-g^{\text{net}}\big((L_c+1)-t\big), & t\le L_c+1,\\
(SOE^{\text{CEV,ini}}_e-\varepsilon)+\delta_{\text{idle}}\,(n_{\text{day}}+1-t), & t>L_c+1,
\end{cases}
\end{equation}
clamped to $[\underline{SOE},\overline{SOE}]$; makes the hard terminal recursively feasible.

%======================================================================
\section{Two-phase / overnight}
Phase 1 is the daytime MILP above. Phase 2 (18:00--08:00) is a deterministic cheapest-hours
recharge returning the MCS to $SOE^{\text{MCS,ini}}$; both batteries are energy-neutral over
the full 24 h cycle.
